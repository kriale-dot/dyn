# ============================================================
# SYN - ETAPA 80
# EXEMPLO DE CONFIGURAÇÃO DE E-MAIL
# ============================================================
#
# NÃO substitua seu .env inteiro.
# Acrescente somente as variáveis abaixo.
#
# DEVELOPMENT sem envio real:
#
# MAIL_TRANSPORT=log
#
# O link será escrito no log do PHP e o token de teste continuará
# disponível na resposta de development.
#
# ------------------------------------------------------------
# SMTP REAL
# ------------------------------------------------------------

APP_WEB_URL=http://localhost:5173

MAIL_TRANSPORT=smtp
MAIL_HOST=mail.exemplo.com.br
MAIL_PORT=587
MAIL_SMTP_AUTH=true
MAIL_USERNAME=usuario@exemplo.com.br
MAIL_PASSWORD=COLOQUE_A_SENHA_NO_ENV
MAIL_ENCRYPTION=tls

MAIL_FROM_ADDRESS=nao-responda@exemplo.com.br
MAIL_FROM_NAME=SYN

# Exemplos de criptografia:
#
# porta 587:
# MAIL_ENCRYPTION=tls
#
# porta 465:
# MAIL_ENCRYPTION=ssl
#
# servidor local sem TLS, somente se realmente apropriado:
# MAIL_ENCRYPTION=none
#
